  // An EMPTY abstract is a MISS, not a hit with no text. The caller drops the trial either way, and
  // conflating the two is exactly how a trial ends up named with nothing behind it.
  const usable = (Array.isArray(rows) ? rows : [])
    .filter((r) => r && r.pmid && typeof r.abstract === "string" && r.abstract.trim().length > 0);

  const trials = [];
  const overBudget = [];
  let spent = 0;
  for (const r of usable) {
    const truncated = r.abstract.length > TRIAL_ABSTRACT_CHARS;
    const abstract = truncated
      ? r.abstract.slice(0, TRIAL_ABSTRACT_CHARS) + " […abstract truncated]"
      : r.abstract;
    // Budget exhausted: drop the WHOLE trial. It joins `missing`, so the caller never names it — which is
    // the safe outcome. Shortening it here would hand over half an abstract dressed as a full one.
    if (spent + abstract.length > TRIALS_TOTAL_CHARS) { overBudget.push(String(r.pmid)); continue; }
    spent += abstract.length;
    trials.push({
      pmid: String(r.pmid),
      title: r.title || "",
      journal: r.journal || "",
      year: r.year || null,
      doi: r.doi || null,
      url: r.url || null,
      abstract,
      truncated,
    });
  }

  // Misses are REPORTED, not left to be inferred from a short array. A trial dropped for budget is a miss
  // in exactly the same sense as one that is not in the corpus: we do not have usable evidence for it.
  const found = new Set(trials.map((t) => t.pmid));
  return jsonOK({
    trials,
    missing: pmids.filter((x) => !found.has(x)),
    dropped_for_budget: overBudget,
    any_truncated: trials.some((t) => t.truncated),
  }, origin);
}
